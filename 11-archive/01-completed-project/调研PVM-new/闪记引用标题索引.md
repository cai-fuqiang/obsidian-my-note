# 闪记引用标题索引

```dataviewjs
const sourcePath = "01-project/调研PVM/闪记.md";
const content = await dv.io.load(sourcePath);

const rows = [];
const lines = content.split("\n");

for (let i = 0; i < lines.length; i++) {
  const line = lines[i];
  const callout = line.match(/^>\s*\[!PDF(?:\|([^\]]*))?\]/);
  if (!callout) continue;

  const link = line.match(/\[\[([^\]|]*?\.pdf(?:#[^\]|]*)?)(?:\|([^\]]+))?\]\]/);
  if (!link) continue;

  const target = link[1];
  const title = (link[2] ?? target).trim();
  const kind = (callout[1] ?? "").trim() || "-";
  const page = target.match(/[?#&]page=(\d+)/)?.[1] ?? "-";

  rows.push({
    index: rows.length + 1,
    line: i,
    title,
    kind,
    page,
  });
}

function sourceLink(row) {
  const link = document.createElement("a");
  link.textContent = row.title;
  link.href = "#";
  link.onclick = async event => {
    event.preventDefault();
    const file = app.vault.getAbstractFileByPath(sourcePath);
    if (!file) return;
    await app.workspace.getLeaf(false).openFile(file, {
      active: true,
      eState: { line: row.line },
    });
  };
  return link;
}

function rowCells(row) {
  return [
    row.index,
    sourceLink(row),
    row.kind,
    row.page,
  ];
}

function searchableText(row) {
  return [
    row.index,
    row.title,
    row.kind,
    row.page,
  ].join(" ").toLowerCase();
}

const state = { keyword: "", composing: false };
const root = dv.container;

function renderTable(container, rows) {
  const table = container.createEl("table");
  const thead = table.createEl("thead");
  const header = thead.createEl("tr");
  ["序号", "标题", "标记", "页码"].forEach(name => header.createEl("th", { text: name }));

  const tbody = table.createEl("tbody");
  for (const row of rows) {
    const tr = tbody.createEl("tr");
    for (const cell of rowCells(row)) {
      const td = tr.createEl("td");
      if (cell instanceof HTMLElement) {
        td.appendChild(cell);
      } else {
        td.textContent = String(cell);
      }
    }
  }
}

function matchedRows() {
  const keyword = state.keyword.trim().toLowerCase();
  if (!keyword) return rows;

  return rows.filter(row => searchableText(row).includes(keyword));
}

function renderResults(container) {
  container.empty();
  const result = matchedRows();

  if (rows.length === 0) {
    dv.el("p", `没有在 [[${sourcePath}|闪记]] 中找到 PDF 引用标题。`, { container });
  } else if (result.length === 0) {
    dv.el("p", "没有匹配的引用标题。", { container });
  } else {
    dv.el("p", `共 ${result.length} / ${rows.length} 条`, { container });
    renderTable(container, result);
  }
}

function render() {
  root.empty();

  const search = root.createEl("input", {
    type: "search",
    placeholder: "搜索标题 / 标记 / 页码",
    value: state.keyword,
  });
  search.style.width = "100%";
  search.style.marginBottom = "0.75rem";

  const results = root.createDiv();

  search.oncompositionstart = () => {
    state.composing = true;
  };

  search.oncompositionend = () => {
    state.composing = false;
    state.keyword = search.value;
    renderResults(results);
  };

  search.oninput = () => {
    state.keyword = search.value;
    if (!state.composing) renderResults(results);
  };

  renderResults(results);
}

render();
```
