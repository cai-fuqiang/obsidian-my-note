# 执行下面脚本
```sh
cd /sys/kernel/tracing/events/signal/signal_generate
# echo "(sig == SIGABRT) || (sig == SIGBUS) || (sig == SIGKILL) || sig == (SIGSEGV)" > filter
echo "(sig == 6) || (sig == 7) || (sig == 9) || (sig == 11)" > filter
echo 'stacktrace if (sig == 6) || (sig == 7) || (sig == 9) || (sig == 11)' > trigger
echo 1 > enable
cd -
```
# 自测

## 使用 SIGTERM信号
```
# 启动一个sleep进程
[root@localhost ~]# sleep 10000 &
[1] 1286

# 使用 SIGTERM信号杀死
[root@localhost ~]# ps aux |grep sleep
root        1286  0.0  0.0  21508  1980 pts/1    S    01:44   0:00 sleep 10000
root        1288  0.0  0.0  22324  2052 pts/0    S+   01:44   0:00 grep --color=auto sleep
[root@localhost ~]# kill 1286

# 查看trace
```
